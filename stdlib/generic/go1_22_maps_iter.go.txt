package maps

// NOTE(yaegi): Go 1.23+ iterator shapes, ported without the "iter" import
// seam (the interpreter has no iter stdlib symbols). The returned sequence
// uses the underlying func(yield func(K) bool) type, which is identical to
// the underlying type of iter.Seq[K], so values convert implicitly and the
// T7 range-over-func support executes them. Behavior matches upstream
// maps/iter.go for the covered symbols.

// Keys returns an iterator over keys in m.
// The iteration order is not specified and is not guaranteed
// to be the same from one call to the next.
func Keys[Map ~map[K]V, K comparable, V any](m Map) func(yield func(K) bool) {
	return func(yield func(K) bool) {
		for k := range m {
			if !yield(k) {
				return
			}
		}
	}
}
